資料與 Supabase

不設定也會自動存在這台瀏覽器。設定 Supabase 後可跨裝置安全合併，同一筆資料以各欄位最新時間為準，不用手動判斷上傳或下載。

尚未設定雲端。
第一次建立 Supabase 資料表
create table if not exists public.user_vocab_data (
  user_id uuid primary key references auth.users(id) on delete cascade,
  payload jsonb not null default '{}'::jsonb,
  updated_at timestamptz not null default now()
);
alter table public.user_vocab_data enable row level security;
create policy "own row select" on public.user_vocab_data for select using (auth.uid() = user_id);
create policy "own row insert" on public.user_vocab_data for insert with check (auth.uid() = user_id);
create policy "own row update" on public.user_vocab_data for update using (auth.uid() = user_id) with check (auth.uid() = user_id);
create policy "own row delete" on public.user_vocab_data for delete using (auth.uid() = user_id);
