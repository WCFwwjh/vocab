Supabase 第一次設定方式

建立專案後，在 SQL Editor 執行以下 SQL，再把 Project URL 與 anon key 貼到上方：

create table public.user_vocab_data (
  user_id uuid primary key references auth.users(id) on delete cascade,
  payload jsonb not null default '{}'::jsonb,
  updated_at timestamptz not null default now()
);
alter table public.user_vocab_data enable row level security;
create policy "read own data" on public.user_vocab_data for select using (auth.uid() = user_id);
create policy "insert own data" on public.user_vocab_data for insert with check (auth.uid() = user_id);
create policy "update own data" on public.user_vocab_data for update using (auth.uid() = user_id);
